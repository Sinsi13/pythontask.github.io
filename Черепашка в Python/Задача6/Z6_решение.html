import turtle as t
t.tracer(0)
t.left(90)
t.screensize(10000, 10000)
k = 30
for i in range(10):
t.forward(22*k)
t.right(90)
t.forward(16*k)
t.right(90)
t.up()
t.forward(1*k)
t.right(90)
t.forward(1*k)
t.left(90)
t.down()
for i in range(10):
t.forward(72*k)
t.right(90)
t.forward(79*k)
t.right(90)
t.up()
for x in range(0, 81):
for y in range(0, 74):
t.goto(x*k, y*k)
t.dot(4, "red")
t.update()
t.done()
